Commit message

Draft a Conventional Commit message for the staged changes. The model may only run git; the result is validated deterministically and retried until it passes.


cd your-repo && git add -p
spl path/to/commit-message.spl


let TYPES = ['feat', 'fix', 'docs', 'refactor', 'test', 'chore', 'perf']

def is_valid(msg) {
    let first = msg.splitlines()[0]
    if len(first) > 72 or ': ' not in first {
        return False
    }
    return first.split(':')[0].split('(')[0] in TYPES
}

let msg = ''
let attempt = 0

restrict exec 'git' {
    forbid 'wrapping the message in quotes or code fences' {
        while attempt < 3 {
            let attempt += 1
            let msg = shell{command='git diff --staged'} '''
Write a Conventional Commit message for this diff: a subject line of at most 72 characters
("type(scope): summary", type is one of {TYPES}), then a blank line and a short body.
Reply with ONLY the message. If the diff is empty reply exactly NO_CHANGES.'''
            if msg == 'NO_CHANGES' {
                fail 'nothing is staged'
            }
            if is_valid(msg) {
                break
            }
            log("attempt {attempt}: invalid message, retrying")
        }
    }
}

assert is_valid(msg) 'could not produce a valid commit message'
success msg
🔑
Pattern: deterministic validator

The model proposes; is_valid() decides. The loop gives it three tries, and the final assert makes sure a script never succeeds with an invalid message.

Triage issues

Classify a batch of issues into labels and priorities. LLM output is treated as untrusted input: deterministic code checks it and asks again, telling the model why it was rejected.


let LABELS = ['bug', 'feature', 'question', 'docs']
let PRIORITIES = ['low', 'medium', 'high']

let issues = [
    {'id': 101, 'title': 'App crashes on startup', 'body': 'Since 2.3 the app exits with a segfault right after the splash screen on macOS.'},
    {'id': 102, 'title': 'Support dark mode', 'body': 'It would be great to have a dark theme, my eyes hurt at night.'},
    {'id': 103, 'title': 'How do I export to CSV?', 'body': 'I cannot find the export option anywhere in the menus.'},
    {'id': 104, 'title': 'Typo in install guide', 'body': 'The guide says "npm instal", should be "npm install".'},
]

def triage(issue) {
    let problem = ''
    for attempt in range(3) {
        try {
            let r = ?{fmt='object'} '''
Classify this GitHub issue. {problem}
Title: {issue['title']}
Body: {issue['body']}

Reply as JSON: {{"label": one of {LABELS}, "priority": one of {PRIORITIES}, "reason": "one short sentence"}}'''
            assert r.label in LABELS "label must be one of {LABELS}"
            assert r.priority in PRIORITIES "priority must be one of {PRIORITIES}"
            return {'id': issue['id'], 'title': issue['title'], 'label': r.label, 'priority': r.priority, 'reason': r.reason}
        } except err {
            let problem = "Your previous answer was rejected ({err.reason}). Follow the format exactly."
            log("issue {issue['id']}: attempt {attempt + 1} failed: {err.reason}")
        }
    }
    fail "could not triage issue {issue['id']}"
}

let results = []
let counts = {}
for issue in issues {
    let t = triage(issue)
    results.append(t)
    if t['label'] in counts {
        let counts[t['label']] += 1
    } else {
        let counts[t['label']] = 1
    }
}

let urgent = []
for t in results {
    if t['priority'] == 'high' {
        urgent.append(t['title'])
    }
}

let summary = ?{len='sentence'} 'Summarize this triage result for the team channel: {counts}. High priority: {urgent}.'
success {'counts': counts, 'urgent': urgent, 'results': results, 'summary': summary}

Extract an invoice

Extract structured data from messy text and cross-check it with plain arithmetic. If the numbers do not add up, the rejection reason is fed back to the model for another try.


let document = '''
ACME Industrial Supplies GmbH            Invoice No. INV-2026-0042
Date: 2026-03-14                          Currency: EUR

  3 x Hex bolt M8 (box of 100)        @ 12.50     37.50
  2 x Safety gloves, size L           @  8.90     17.80
  1 x Cordless drill                  @ 119.00   119.00

Subtotal 174.30   VAT 19%  33.12   TOTAL DUE 207.42
'''

let feedback = ''
let invoice = None

for attempt in range(3) {
    if invoice != None {
        break
    }
    try {
        let r = ?{fmt='object'} '''
Extract the invoice below. {feedback}
Reply as JSON: {{"vendor": str, "number": str, "date": "YYYY-MM-DD", "currency": str,
"lines": [{{"description": str, "quantity": number, "unit_price": number}}], "subtotal": number, "total": number}}

{document}'''
        assert type(r.lines) == 'list' and len(r.lines) > 0 'lines must be a non-empty list'
        let computed = 0
        for line in r.lines {
            let computed += line.quantity * line.unit_price
        }
        assert abs(computed - r.subtotal) < 0.01 "lines add up to {round(computed, 2)} but subtotal is {r.subtotal}"
        assert r.total >= r.subtotal "total {r.total} is smaller than subtotal {r.subtotal}"
        let invoice = r
    } except err {
        let feedback = "Your previous extraction was rejected: {err.reason}. Re-read the document and fix it."
        log("attempt {attempt + 1} rejected: {err.reason}")
    }
}

assert invoice != None 'extraction kept failing validation'
success invoice

Code review as a CI gate

A two-stage review: think deeply in prose, then structure the findings. The review phase may only touch ./src; the report is written under ./reports. Exit code 1 (failure payload on stdout) when high-severity findings exist.


spl examples/code-review.spl --cwd <project>


let target = './src/restrict.ts'

let notes = ''
restrict './src' {
    forbid 'suggesting purely stylistic changes' {
        let notes = think{level='high'} '''
Read {target} and review it for security problems and correctness bugs.
List each finding with its location, why it matters and a suggested fix.'''
    }
}

let review = ?{fmt='object'} '''
Convert this review into JSON:
{{"findings": [{{"severity": "low"|"medium"|"high", "title": str, "location": str, "fix": str}}]}}
Use an empty list when there are no findings.

{notes}'''

let lines = ["# Review of {target}", '']
let high = 0
for f in review.findings {
    assert f.severity in ['low', 'medium', 'high'] "bad severity '{f.severity}'"
    if f.severity == 'high' {
        let high += 1
    }
    lines.append("- **{f.severity.upper()}** {f.title} ({f.location})")
    lines.append("  - Fix: {f.fix}")
}
lines.append('')
lines.append("{len(review.findings)} finding(s), {high} high severity")

restrict './reports' {
    write_file{path='./reports/review.md', content='\n'.join(lines)} 'Confirm that the report was written'
}

if high > 0 {
    fail "{high} high severity finding(s), see ./reports/review.md"
}
success '\n'.join(lines)

Release notes

Generate release notes from recent git history and save them to ./out/RELEASE_NOTES.md. The model may only run git; file access is limited to ./out. Note how rules and restrictions nest.


let notes = ''

restrict exec 'git' {
    forbid 'inventing changes that are not in the commit list' {
        enforce 'Group entries under: Features, Fixes, Other. Omit empty groups. Plain Markdown, no preamble' {
            let notes = shell{command='git log --oneline -n 40'} '''
Write release notes for end users from these commits (one per line: hash and subject).
Rewrite technical subjects into clear user-facing sentences and drop merge/chore noise.'''
        }
    }
}

assert len(notes) > 20 'release notes are suspiciously short'

restrict './out' {
    write_file{path='./out/RELEASE_NOTES.md', content=notes} 'Confirm that the file was written'
}
success notes

Web digest

Summarize a web page with network access limited to a single host — and prove the sandbox holds by catching a blocked request to a second host.


restrict net 'news.ycombinator.com' {
    let headlines = web_fetch{url='https://news.ycombinator.com'} '''
From this HTML list the 10 top stories as a Markdown bullet list: "- title (domain)".
Reply with ONLY the list.'''

    let themes = ?{len='paragraph'} 'What themes run through these headlines?\n{headlines}'

    try {
        web_fetch{url='https://example.org'} 'Return the page'
        fail 'the request to a second host should have been blocked'
    } except err {
        assert err.status_code == 'restriction_violation' "unexpected failure: {err.reason}"
        log("blocked as expected: {err.reason}")
    }
}

success "{headlines}\n\n{themes}"

Fix failing tests

An agent loop: run the suite, let the model repair the source, repeat (at most three rounds). Only npm may be executed, files are limited to the project, and tests are off limits. Every npm test run costs one LLM call, because tool output is passed through your instructions.


spl examples/fix-failing-tests.spl --cwd <your npm project>


let attempt = 0
let green = False

restrict exec 'npm' {
    restrict '.' {
        forbid 'editing or deleting test files, or weakening assertions' {
            while attempt < 3 and not green {
                let attempt += 1
                try {
                    shell{command='npm test'} 'Reply with OK'
                    let green = True
                } except err {
                    log("tests failing (round {attempt}): {err.status_code}")
                    think{level='medium'} '''
The test run failed with this output:
{err.reason}

Find the bug in the source code (not the tests), fix it with the file tools, and describe the change in one sentence.'''
                }
            }
        }
    }
}

assert green 'tests are still failing after 3 rounds'
success "tests pass after {attempt} run(s)"
🔑
Pattern: failure as input

A failing npm test is a tool_error whose reason contains the test output. Catching it turns the failure into the prompt for the next step.

Translate docs

Translate a document into several languages, one file each, without leaving the project folder. The model reads the source and writes each output itself — the output path is pinned by the script.


let locales = {'de': 'German', 'pl': 'Polish', 'ja': 'Japanese'}
let written = []

restrict '.' {
    enforce 'Keep Markdown formatting, URLs and code identifiers untranslated' {
        for code, language in items(locales) {
            write_file{path="./out/translations/product-update.{code}.md"} '''
Read ./examples/data/product-update.md, translate it to {language}, and write the translation to the file.'''
            written.append("./out/translations/product-update.{code}.md")
        }
    }
}

success written

Journal with persistent state

A rolling journal that remembers previous runs through the variable store. Top-level variables are saved on exit and reloaded on the next run, so state survives between runs.


spl examples/journal.spl --store journal.json


try {
    runs
} except first_run {
    # `runs` is not in the store yet, so this is the first run (the caught payload is dropped again)
    let first_run = None
    let runs = 0
    let history = []
}

let runs += 1

let todays = read_file{path='./examples/data/notes.txt'} 'Summarize these notes in one sentence'
history.append({'run': runs, 'summary': todays})
if len(history) > 5 {
    let history = history[1:]
}

let trend = ''
if len(history) > 1 {
    let trend = ?{len='paragraph'} '''
Here are my last journal summaries, oldest first: {history}
What is changing from run to run? Be concrete.'''
}

success {'run': runs, 'today': todays, 'trend': trend}

Feature tours

Small scripts that exercise one area each.

basics.spl — offline-friendly tour


@{model='gpt-5'}

# Deterministic part: functions, loops, interpolation
def fib(n) {
    if n < 2 {
        return n
    }
    return fib(n - 1) + fib(n - 2)
}

let nums = []
for i in range(8) {
    nums.append(fib(i))
}
log("fib: {nums}")

# LLM call with modifiers and parsed object
let joke = ?{len='sentence', fmt='object'} 'A joke about programmers as {{"setup": ..., "punchline": ...}}'
log("setup: {joke['setup']}")

# Scoped constraints
enforce 'Answer in English' {
    forbid 'Mentioning the word cat' {
        let animal = ?{len='word'} 'Name a pet animal'
    }
}

# Failure handling
try {
    assert len(nums) == 3 'expected three numbers'
} except err {
    log("caught {err.status_code}: {err.reason}")
}

success {'fib': nums, 'animal': animal}
